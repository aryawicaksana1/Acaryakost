import { Resend } from 'resend';

const resend = new Resend('re_123456789');

resend.emails.send({
  from: 'onboarding@resend.dev',
  to: 'steve@apple.com',
  subject: 'Hello World',
  html: '<p>Congrats on sending your <strong>first email</strong>!</p>'
});

Successfully sent to steve@apple.com in 143ms.
